cmake_minimum_required(VERSION 3.16)
project(dmrecomp C)

set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# The front end: Win32 on Windows, SDL2 everywhere else. -DDM_SDL=ON builds the SDL2 one on Windows too.
if(WIN32)
  option(DM_SDL "Use the SDL2 front end instead of the Win32 one" OFF)
else()
  set(DM_SDL ON)
endif()

file(GLOB GEN_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/gen/*.c)
file(GLOB RT_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/runtime/*.c)
list(FILTER RT_SRC EXCLUDE REGEX "/plat_[a-z0-9]+\\.c$")
if(DM_SDL)
  list(APPEND RT_SRC ${CMAKE_SOURCE_DIR}/runtime/plat_sdl.c)
else()
  list(APPEND RT_SRC ${CMAKE_SOURCE_DIR}/runtime/plat_win32.c)
endif()
if(NOT GEN_SRC)
  message(FATAL_ERROR "gen/ is empty: run tools/regen.sh <extracted disc dir> first")
endif()

add_executable(dmrecomp ${RT_SRC} ${GEN_SRC})
target_include_directories(dmrecomp PRIVATE runtime gen)
# generated code is huge; -O1 keeps build times sane while staying fast
set_source_files_properties(${GEN_SRC} PROPERTIES COMPILE_OPTIONS "-O1;-fno-strict-aliasing;-w")
target_compile_options(dmrecomp PRIVATE -Wall -Wno-unused-function -fno-strict-aliasing)

if(DM_SDL)
  find_package(SDL2 REQUIRED)
  target_compile_definitions(dmrecomp PRIVATE DM_SDL)
  if(TARGET SDL2::SDL2)
    target_link_libraries(dmrecomp PRIVATE SDL2::SDL2)
  else()   # older SDL2 packages only set variables
    target_include_directories(dmrecomp PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(dmrecomp PRIVATE ${SDL2_LIBRARIES})
  endif()
endif()
if(WIN32)
  target_link_libraries(dmrecomp PRIVATE winmm opengl32 gdi32)
else()
  set(OpenGL_GL_PREFERENCE GLVND)
  find_package(OpenGL REQUIRED)
  set(THREADS_PREFER_PTHREAD_FLAG ON)
  find_package(Threads REQUIRED)
  target_link_libraries(dmrecomp PRIVATE OpenGL::GL Threads::Threads m)
endif()
if(MINGW)
  # the recompiled code keeps the game's call depth on the native stack (runtime/main.c does the same elsewhere)
  target_link_options(dmrecomp PRIVATE -Wl,--stack,16777216)
  if(NOT DM_SDL)
    target_link_options(dmrecomp PRIVATE -static)
  endif()
endif()
